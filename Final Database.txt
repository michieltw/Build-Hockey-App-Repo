// Core users and persons
Table users {
  id integer [pk, increment, note: 'readonly']
  username varchar(100) [unique, not null]
  email varchar(255) [unique, not null, note: 'display as email input']
  password_hash varchar [not null, note: 'readonly']
  role_id integer [ref: > roles.id]
  public_id varchar [unique]
  vip_tier vip_tier [default: 'standard']
  created_at timestamp [default: `now()`, note: 'readonly']
  updated_at timestamp [note: 'readonly']
}

Table persons {
  id integer [pk, increment, note: 'readonly']
  first_name varchar(100) [not null]
  last_name varchar(100) [not null]
  date_of_birth date
  nationality_id integer [ref: > nationalities.id]
  bio text [note: 'display as textarea']
  created_at timestamp [default: `now()`, note: 'readonly']
}

Table roles {
  id integer [pk, increment, note: 'readonly']
  name varchar [unique, not null]
  description text [note: 'display as textarea']
}

Table nationalities {
  id integer [pk, increment, note: 'readonly']
  name varchar [unique, not null]
  code varchar(2) [unique]
}

Table ruling_bodies {

}

Table calendar_events {
  id integer [pk, increment, note: 'readonly']
  title varchar [not null]
  description text [note: 'display as textarea']
  event_type varchar [not null]
  event_date timestamp [not null]
  end_time timestamp
  location varchar
  created_by_user_id integer [ref: > users.id, not null]
  team_id integer [ref: > teams.id]
  game_id integer [ref: > games.id]
  practice_id integer [ref: > practices.id]
  generic_event_id integer [ref: > events.id]
  recurrence_pattern varchar
  status varchar [default: 'scheduled'] [note: 'display as dropdown']
  created_at timestamp [default: `now()`, note: 'readonly']
  updated_at timestamp [note: 'readonly']
}

Table calendar_attendees {
  id integer [pk, increment, note: 'readonly']
  calendar_event_id integer [ref: > calendar_events.id, not null]
  user_id integer [ref: > users.id, not null]
  attendance_status varchar [default: 'pending']
  created_at timestamp [default: `now()`, note: 'readonly']
  indexes {
    (calendar_event_id, user_id) [unique]
  }
}
